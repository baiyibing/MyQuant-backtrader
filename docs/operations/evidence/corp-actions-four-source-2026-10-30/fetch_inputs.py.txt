def fetch(item):
    name, repo, ref, remote_path = item
    path = OUT / name
    if not path.is_file():
        command = (["git", "show", f"{ref}:{remote_path}"] if repo == "MyQuant-backtrader" else
                   ["gh", "api", f"repos/baiyibing/{repo}/contents/{remote_path}?ref={ref}",
                    "-H", "Accept: application/vnd.github.raw+json"])
        result = subprocess.run(command, capture_output=True, timeout=90, check=True)
        path.write_bytes(result.stdout)
    data = path.read_bytes()
    return {"file": name, "repository": repo, "ref": ref, "path": remote_path,
            "bytes": len(data), "sha256": hashlib.sha256(data).hexdigest()}


with ThreadPoolExecutor(max_workers=3) as executor:
    manifest = list(executor.map(fetch, FILES))
(OUT / "manifest.json").write_text(json.dumps(manifest, indent=2) + "\n", encoding="utf-8")
print(json.dumps(manifest, indent=2), flush=True)
