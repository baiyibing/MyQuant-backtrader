"""Offline four-source diagnostics; never publishes data or accepts events."""

from collections import Counter, defaultdict
import csv
from decimal import Decimal, ROUND_HALF_UP
import hashlib
import json
from pathlib import Path
import random
import sys

import pandas as pd
import pyarrow.parquet as pq

HERE = Path(__file__).resolve().parent
REPO = HERE.parents[1]
sys.path.insert(0, str(REPO))
from common.infra.data_root import resolve_parquet_container

INPUTS = HERE / "inputs"
OUT = HERE / "results"
OUT.mkdir(exist_ok=True)
START, END = "2024-01-01", "2026-10-06"


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def read_csv(path):
    with path.open(encoding="utf-8-sig", newline="") as stream:
        return list(csv.DictReader(stream))


def write_csv(name, rows):
    if not rows:
        return
    with (OUT / name).open("w", encoding="utf-8", newline="") as stream:
        writer = csv.DictWriter(stream, fieldnames=list(rows[0]), lineterminator="\n")
        writer.writeheader()
        writer.writerows(rows)


def write_json(name, value):
    (OUT / name).write_text(json.dumps(value, ensure_ascii=False, indent=2, default=str) + "\n", encoding="utf-8")


def amount(value):
    if value is None or str(value).strip() in ("", "None", "nan"):
        return None
    number = Decimal(str(value))
    if not number.is_finite() or number < 0:
        raise ValueError("Invalid source amount")
    return number


def compare(left, right):
    if left is None or right is None:
        return "UNKNOWN"
    return "EXACT" if left == right else "DIFFERENT"


def board(code):
    if code.endswith(".BJ"):
        return "BJ"
    if code.startswith("68") and code.endswith(".SH"):
        return "STAR"
    if code.startswith("60") and code.endswith(".SH"):
        return "SH_MAIN"
    if code.startswith("30") and code.endswith(".SZ"):
        return "CHINEXT"
    if code.startswith("00") and code.endswith(".SZ"):
        return "SZ_MAIN"
    return "OTHER"


manifest = json.loads((INPUTS / "manifest.json").read_text(encoding="utf-8"))
for entry in manifest:
    assert digest(INPUTS / entry["file"]) == entry["sha256"], entry["file"]
wind = read_csv(INPUTS / "wind.csv")
assert digest(INPUTS / "wind.csv") == "6afe807a996fea68f2f34fa43344f44ca8f4254fb8580c9213bef6d443c3ba4d"
cninfo = read_csv(INPUTS / "cninfo_events.csv")
previous = {r["event_id"]: r for r in read_csv(INPUTS / "cninfo_events_v8.csv")}
accepted = {r["event_id"]: r for r in read_csv(INPUTS / "cninfo_accepted.csv")}
sources = {r["announcement_id"]: r for r in json.loads((INPUTS / "cninfo_sources.json").read_text(encoding="utf-8"))}
universe = set((INPUTS / "sample_universe.txt").read_text(encoding="utf-8").split())
hpath = REPO / "_agent_runs/pr531_lake_audit_20261029/hithink_dump.parquet"
assert digest(hpath) == "db19162144ca902160931a9eb9472a56e81580dc6d0f4ff4131b9647971c5a27"
hithink = pq.ParquetFile(hpath).read().to_pandas().drop_duplicates()
hithink["ex_date"] = pd.to_datetime(hithink.ex_date_ms, unit="ms", utc=True).dt.tz_convert("Asia/Shanghai").dt.strftime("%Y-%m-%d")
root = resolve_parquet_container()
qpath = INPUTS / "qmt_ex_date_index.parquet"
qhash = digest(qpath)
assert qhash == "f19177339468ac43372d4298e9d08db9742d5c9e4aacde7afec042e5e522c601"
qmt = pq.ParquetFile(qpath).read().to_pandas()
assert digest(qpath) == qhash, "QMT index changed while reading"
qkeys = {(r.stock_code, f"{r.ex_date[:4]}-{r.ex_date[4:6]}-{r.ex_date[6:8]}"): r.dr for r in qmt.itertuples()}
wind_index, hi_index = defaultdict(list), defaultdict(list)
for row in wind:
    if row["code"] in universe and START <= row["ex_date"] <= END:
        wind_index[(row["code"], row["ex_date"])].append(row)
for row in hithink.to_dict("records"):
    hi_index[(row["thscode"], row["ex_date"])].append(row)
selected = [r for r in cninfo if r["code"] in universe and START <= r["ex_date"] <= END]
counts, policies, rows, differences = Counter(), Counter(), [], []
for row in selected:
    key = row["code"], row["ex_date"]
    hi, wr = hi_index.get(key, []), wind_index.get(key, [])
    old = previous[row["event_id"]]
    hi_cash = sum((amount(r["dividend_per_share"]) for r in hi), Decimal(0)) if hi else None
    hi_bonus = sum((amount(r["per_share_bonus"]) for r in hi), Decimal(0)) if hi else None
    wind_cash_values = [amount(r["cash_div_per_share"]) for r in wr]
    wind_cash = (sum(wind_cash_values, Decimal(0)) if wr and
                 all(value is not None for value in wind_cash_values) else None)
    wind_bonus = sum(((amount(r["bonus_ratio"]) or Decimal(0)) + (amount(r["transfer_ratio"]) or Decimal(0)) for r in wr), Decimal(0)) if wr else None
    cn_cash = amount(row["cash_div_per_share"])
    raw_legs = [amount(row[k]) for k in ("bonus_ratio", "transfer_ratio")]
    raw_bonus = sum(raw_legs, Decimal(0)) if all(x is not None for x in raw_legs) else None
    adapted = accepted.get(row["event_id"])
    adapted_bonus = (sum(((amount(adapted[k]) or Decimal(0)) for k in ("bonus_ratio", "transfer_ratio")), Decimal(0))
                     if adapted else None)
    listing_policy = [k for k in ("bonus_list_date", "transfer_list_date")
                      if not old[k] and row[k] and row[k] == row["ex_date"]]
    blank_policy = [k for k in ("bonus_ratio", "transfer_ratio")
                    if adapted and not row[k] and amount(adapted[k]) in (None, Decimal(0))]
    policies.update({"listing_ex_date_fields": len(listing_policy), "blank_to_zero_fields": len(blank_policy)})
    counts["listing_policy_events"] += bool(listing_policy)
    counts["blank_to_zero_events"] += bool(blank_policy)
    result = dict(code=row["code"], ex_date=row["ex_date"], event_id=row["event_id"],
                  announcement_id=row["announcement_id"], announcement_url=sources[row["announcement_id"]]["url"],
                  pdf_sha256=sources[row["announcement_id"]]["pdf_sha256"],
                  cninfo_status=row["status"], cninfo_reasons=row["reasons"], adapter_accepted=bool(adapted),
                  cninfo_cash=row["cash_div_per_share"], cninfo_bonus_raw=row["bonus_ratio"],
                  cninfo_transfer_raw=row["transfer_ratio"],
                  cninfo_record_date=row["record_date"], cninfo_pay_date=row["pay_date"],
                  cninfo_bonus_list_date=row["bonus_list_date"], cninfo_transfer_list_date=row["transfer_list_date"],
                  listing_policy_fields="|".join(listing_policy), adapter_zero_policy_fields="|".join(blank_policy),
                  hithink_rows=len(hi), hithink_cash=hi_cash, hithink_bonus_total=hi_bonus,
                  wind_rows=len(wr), wind_cash=wind_cash, wind_bonus_total=wind_bonus,
                  qmt_date_present=key in qkeys, qmt_amounts="NOT_PERSISTED_IN_INDEX",
                  hi_vs_wind_cash=compare(hi_cash, wind_cash), hi_vs_wind_bonus=compare(hi_bonus, wind_bonus),
                  hi_vs_cninfo_cash=compare(hi_cash, cn_cash), hi_vs_cninfo_bonus_raw=compare(hi_bonus, raw_bonus),
                  hi_vs_cninfo_bonus_adapted=compare(hi_bonus, adapted_bonus) if adapted else "NOT_ACCEPTED",
                  cash_round4_equal_diagnostic=bool(hi_cash is not None and cn_cash is not None and
                      hi_cash != cn_cash and hi_cash.quantize(Decimal("0.0001"), rounding=ROUND_HALF_UP) ==
                      cn_cash.quantize(Decimal("0.0001"), rounding=ROUND_HALF_UP)))
    for column in ("hi_vs_wind_cash", "hi_vs_wind_bonus", "hi_vs_cninfo_cash", "hi_vs_cninfo_bonus_raw", "hi_vs_cninfo_bonus_adapted"):
        counts[column + ":" + result[column]] += 1
    counts["qmt_date_present"] += result["qmt_date_present"]
    counts["hithink_date_present"] += bool(hi)
    counts["wind_date_present"] += bool(wr)
    counts["adapter_accepted"] += bool(adapted)
    rows.append(result)
    if any(result[k] == "DIFFERENT" for k in ("hi_vs_wind_cash", "hi_vs_wind_bonus", "hi_vs_cninfo_cash", "hi_vs_cninfo_bonus_adapted")):
        differences.append(result)
assert len(selected) == 147 and counts["adapter_accepted"] == 115
assert len(cninfo) == 152 and len(accepted) == 118
write_csv("four_source_events.csv", rows)
write_csv("four_source_differences.csv", differences)
write_csv("policy_derived_events.csv", [r for r in rows if r["listing_policy_fields"] or r["adapter_zero_policy_fields"]])
write_json("report.json", dict(scope={"codes": len(universe), "start": START, "end": END,
    "cninfo_events_total": len(cninfo), "cninfo_selected": len(selected), "accepted_total": len(accepted),
    "review_total": len(read_csv(INPUTS / "review_list.csv"))}, metrics=dict(counts), policy_field_counts=dict(policies),
    distinct_difference_events=len(differences), cninfo_wind_run_a=json.loads((INPUTS/"run_a_report.json").read_text(encoding="utf-8"))["class_counts"],
    cninfo_wind_run_b=json.loads((INPUTS/"run_b_report.json").read_text(encoding="utf-8"))["class_counts"],
    safety={"owner_signature": None, "lake_write_authorized": False, "source_switch_authorized": False,
            "automatic_acceptance": False, "hithink_round4_policy_applied": False,
            "qmt_amounts_compared": False, "coverage_claim": "fixed-sample diagnostics only"},
    source_snapshots={"cninfo_v10": "f00c75708f92d93db3ce7583e59b6cadc615c84b",
        "cninfo_evidence": "ce30b8bf456fdaf7639b61650e339671482eaf09", "wind_sha256": digest(INPUTS/"wind.csv"),
        "hithink_sha256": digest(hpath), "qmt_ex_date_index_sha256": qhash,
        "qmt_root": str(root), "qmt_ex_date_max": qmt.ex_date.max(),
        "listing_policy_evidence": "v8 pre-fallback extracted field vs v10; raw PDF text not replayed"}))

# Independent holdout frame: QMT codes with an ex-date since 2024; no hithink filter.
excluded = universe | {"000001.SZ", "000002.SZ"}
frame = set(qmt.loc[qmt.ex_date >= "20240101", "stock_code"]) - excluded
rng = random.Random(20261030)
targets = {"SH_MAIN": 60, "SZ_MAIN": 60, "CHINEXT": 60, "STAR": 40, "BJ": 20}
holdout = []
for tier, target in targets.items():
    candidates = sorted(code for code in frame if board(code) == tier)
    if len(candidates) < target:
        raise ValueError(f"Insufficient independent sampling frame for {tier}")
    holdout.extend({"code": code, "stratum": tier} for code in sorted(rng.sample(candidates, target)))
write_csv("holdout_240.csv", holdout)
write_json("holdout_manifest.json", dict(seed=20261030, sample_count=len(holdout), strata=targets,
    frame="QMT ex_date_index codes with ex_date>=20240101; independent of hithink and Wind availability",
    qmt_snapshot_sha256=qhash,
    exclusions=sorted(excluded), implementation="extractor mvp-v10 and current approved policies; frozen",
    historical_ex_date_window=[START, END], incremental_ex_date_window=["2026-10-07", "2026-10-28"],
    announcement_query_window=["2023-12-01", "2026-10-28"],
    sample_sha256=digest(OUT/"holdout_240.csv"), lake_write_authorized=False, owner_signature=None))
print((OUT / "report.json").read_text(encoding="utf-8"), flush=True)
print("Independent holdout frozen: 240 codes, seed 20261030.", flush=True)
