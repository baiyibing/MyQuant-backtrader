import csv

import pandas as pd
import pyarrow.parquet as pq

HERE = Path(__file__).resolve().parent
RAW = HERE / "bt_corp_actions_cninfo_backup_20261030"

def source(aid):
    entry = json.loads((RAW / "events" / f"{aid}.json").read_text(encoding="utf-8"))
    text = (RAW / "txt" / f"{aid}.txt").read_text(encoding="utf-8")
    assert hashlib.sha256(text.encode("utf-8")).hexdigest() == entry["source"]["text_sha256"]
    assert hashlib.sha256((RAW / "pdf" / f"{aid}.pdf").read_bytes()).hexdigest() == entry["source"]["pdf_sha256"]
    return entry, text

hframe = pq.ParquetFile(hpath).read().to_pandas().drop_duplicates()
hframe["day"] = pd.to_datetime(hframe.ex_date_ms, unit="ms", utc=True).dt.tz_convert("Asia/Shanghai").dt.strftime("%Y-%m-%d")
for day, expected in (("2024-07-11", "0.0999985"), ("2025-07-18", "0.0499999")):
    selected = hframe[(hframe.thscode == "300243.SZ") & (hframe.day == day)]
    assert len(selected) == 1 and str(selected.iloc[0].dividend_per_share) == expected
