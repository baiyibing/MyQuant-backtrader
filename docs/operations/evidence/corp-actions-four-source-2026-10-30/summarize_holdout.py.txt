def read_csv(path):
    with path.open(encoding="utf-8", newline="") as stream:
        return list(csv.DictReader(stream))


def dec(value):
    return None if value is None or str(value) == "" else Decimal(str(value))


def total(rows, fields, *, blank_zero=False):
    values = [dec(row[field]) for row in rows for field in fields]
    if not rows or (not blank_zero and any(value is None for value in values)):
        return None
    return sum((value if value is not None else Decimal(0) for value in values), Decimal(0))


def comparison(left, right):
    return "UNKNOWN" if left is None or right is None else "EXACT" if left == right else "DIFFERENT"


def single_date(rows, field):
    values = {row[field] for row in rows}
    return next(iter(values)) if len(values) == 1 and "" not in values else None


def csv_out(name, rows):
    if not rows:
        return
    with (OUT / name).open("w", encoding="utf-8", newline="") as stream:
        writer = csv.DictWriter(stream, fieldnames=list(rows[0]), lineterminator="\n")
        writer.writeheader()
        writer.writerows(rows)


def save(name, value):
    (OUT / name).write_text(json.dumps(value, ensure_ascii=False, indent=2, default=str) + "\n", encoding="utf-8")


def load_module():
    spec = importlib.util.spec_from_file_location("holdout_v10_observed", HERE / "inputs/extractor_v10.py")
    obj = importlib.util.module_from_spec(spec)
    sys.modules[spec.name] = obj
    spec.loader.exec_module(obj)
    return obj


def observe(module, text, row):
    original_fill, original_resolve = module._fill_missing_share_listing_dates, module._resolve_distribution_values
    trace, before, depth = [], {}, 0
    fields = iter(("cash_div_per_share", "bonus_ratio", "transfer_ratio"))

    def fill(values, reasons):
        before.update(values)
        return original_fill(values, reasons)

    def resolve(values, adjusted, original, **kwargs):
        nonlocal depth
        outer = depth == 0
        field = next(fields) if outer else None
        depth += 1
        try:
            chosen = original_resolve(values, adjusted, original, **kwargs)
        finally:
            depth -= 1
        if outer:
            trace.append(dict(field=field, extracted_values=sorted(map(str, values)),
                              adjusted_values=sorted(map(str, adjusted)), original_values=sorted(map(str, original)),
                              actual_values=sorted(map(str, kwargs.get("actual") or [])), chosen=chosen))
        return chosen

    module._fill_missing_share_listing_dates, module._resolve_distribution_values = fill, resolve
    try:
        result = module.extract_corp_actions(text, title=row["title"])
    finally:
        module._fill_missing_share_listing_dates, module._resolve_distribution_values = original_fill, original_resolve
    assert result.status == row["status"]
    assert all((result.fields[key] or "") == row[key] for key in module.FIELDS), row["announcement_id"]
    listing = [key for key in ("bonus_list_date", "transfer_list_date")
               if not before.get(key) and result.fields.get(key)]
    return {"event_id": row["event_id"], "announcement_id": row["announcement_id"], "code": row["code"],
            "ex_date": row["ex_date"], "amount_selection": trace, "before_listing_policy": before,
            "listing_policy_fields": listing,
            "adapter_zero_policy_fields": [key for key in ("bonus_ratio", "transfer_ratio")
                if row["adapter_accepted"] == "True" and not row[key]],
            "method": "Observational wrappers call unmodified v10 functions; result fields/status asserted unchanged"}
